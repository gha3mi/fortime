   subroutine run_test1(ut)
      type(unit_test), intent(inout) :: ut
      type(timer) :: t

      call t%timer_start()
      call sleep(1)
      call t%timer_stop()

      call ut%check(res=t%elapsed_time, expected=1.0_rk, tol=1.0e-1_rk, msg='test1')
   end subroutine run_test1
